const SHEET_NAME = 'Dados'; function doPost(e) { try { const body = JSON.parse(e.postData.contents); const ss = SpreadsheetApp.openById(body.sheetId); const sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME); const { user, key, value } = body; const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] === user && rows[i][1] === key) { sheet.getRange(i + 1, 3).setValue(value); sheet.getRange(i + 1, 4).setValue(new Date().toISOString()); return ok('updated'); } } sheet.appendRow([user, key, value, new Date().toISOString()]); return ok('created'); } catch(err) { return err_(err.toString()); } } function doGet(e) { try { const { sheetId, user, key } = e.parameter; const ss = SpreadsheetApp.openById(sheetId); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) return ok(null); const rows = sheet.getDataRange().getValues(); if (key) { for (let i = 1; i < rows.length; i++) { if (rows[i][0] === user && rows[i][1] === key) return ok(rows[i][2]); } return ok(null); } const result = {}; rows.slice(1).filter(r => r[0] === user).forEach(r => result[r[1]] = r[2]); return ok(result); } catch(err) { return err_(err.toString()); } } const ok = v => ContentService.createTextOutput(JSON.stringify({ok:true,value:v})).setMimeType(ContentService.MimeType.JSON); const err_= m => ContentService.createTextOutput(JSON.stringify({ok:false,error:m})).setMimeType(ContentService.MimeType.JSON);
